-- ─── CLICK ENGINE ────────────────────────────────────────────
-- Tries every known method to fire a UI element as a click.
-- Returns true if at least one method fired.
local function FireElement(inst)
    local fired = false

    -- TextButton / ImageButton — most common
    if inst:IsA("GuiButton") then
        -- pcall because some games override these with metamethods
        pcall(function()
            inst.MouseButton1Down:Fire()
            inst.MouseButton1Up:Fire()
            inst.MouseButton1Click:Fire()
        end)
        -- Also try fireproximityprompt style via internal fire
        pcall(function() firesignal(inst.MouseButton1Click) end)
        fired = true
    end

    -- ClickDetector parented inside a GUI (rare but exists)
    if inst:IsA("ClickDetector") then
        pcall(function() fireclickdetector(inst) end)
        fired = true
    end

    -- ProximityPrompt that happens to be in UI (some games do this)
    if inst:IsA("ProximityPrompt") then
        pcall(function() fireproximityprompt(inst) end)
        fired = true
    end

    -- ImageLabel / TextLabel / Frame — not directly clickable,
    -- but we can synthesize a touch on their AbsolutePosition
    if not fired and inst:IsA("GuiObject") then
        pcall(function()
            local pos = inst.AbsolutePosition + inst.AbsoluteSize * 0.5
            -- Simulate a tap at center of element via VirtualInputManager
            game:GetService("VirtualInputManager"):SendMouseButtonEvent(
                pos.X, pos.Y, 0, true,  game, 0)
            game:GetService("VirtualInputManager"):SendMouseButtonEvent(
                pos.X, pos.Y, 0, false, game, 0)
        end)
        fired = true
    end

    return fired
end

-- ─── SCAN ENGINE ─────────────────────────────────────────────
local lastResults   = {}
local hlInstances   = {}   -- {SelectionBox or UIStroke} to clean up

local function ClearHighlights()
    for _, obj in ipairs(hlInstances) do
        pcall(function() obj:Destroy() end)
    end
    hlInstances = {}
end

-- Highlight a GuiObject with a colored UIStroke overlay
local function HighlightUi(inst)
    pcall(function()
        -- Remove any existing stroke we added
        local existing = inst:FindFirstChild("__UiScanHL")
        if existing then existing:Destroy() end

        local stroke = Instance.new("UIStroke")
        stroke.Name      = "__UiScanHL"
        stroke.Color     = Color3.fromRGB(255, 100, 200)
        stroke.Thickness = 3
        stroke.Parent    = inst
        hlInstances[#hlInstances + 1] = stroke
    end)
end

local FilterModes  = {"All", "Buttons", "TextLabels", "Frames", "ImageButtons"}
local FilterColors = {
    All          = Color3.fromRGB(210, 210, 210),
    Buttons      = Color3.fromRGB(255, 170, 80),
    TextLabels   = Color3.fromRGB(100, 200, 255),
    Frames       = Color3.fromRGB(180, 255, 100),
    ImageButtons = Color3.fromRGB(255, 100, 200),
}

local function PassesFilter(inst)
    local mode = PersistState.FilterMode
    if mode == "All" then
        return inst:IsA("GuiObject")
    elseif mode == "Buttons" then
        return inst:IsA("TextButton") or inst:IsA("ImageButton")
    elseif mode == "TextLabels" then
        return inst:IsA("TextLabel")
    elseif mode == "Frames" then
        return inst:IsA("Frame") or inst:IsA("ScrollingFrame")
    elseif mode == "ImageButtons" then
        return inst:IsA("ImageButton")
    end
    return false
end

local function RunSearch(query)
    ClearHighlights()
    lastResults = {}
    if query == "" then return end

    local queryLower = query:lower()

    local roots = {LocalPlayer:WaitForChild("PlayerGui")}
    if PersistState.ScanCoreGui then
        roots[#roots + 1] = game:GetService("CoreGui")
    end

    for _, root in ipairs(roots) do
        for _, inst in ipairs(root:GetDescendants()) do
            if not inst.Name:lower():find(queryLower, 1, true) then continue end
            if not PassesFilter(inst) then continue end
            -- Skip our own GUI
            if inst:IsDescendantOf(ScreenGui) then continue end

            HighlightUi(inst)

            lastResults[#lastResults + 1] = {
                instance = inst,
                name     = inst.Name,
                class    = inst.ClassName,
                text     = (inst:IsA("TextLabel") or inst:IsA("TextButton"))
                            and inst.Text:sub(1, 24) or "",
            }
        end
    end

    return #lastResults
end

-- Continuous auto-click: loops results, re-searches when exhausted
local autoRunning = false

local AutoBtn, AutoLbl = CreateActionButton(
    "Continuous Click: OFF",
    Color3.fromRGB(20, 30, 45),
    Color3.fromRGB(40, 80, 120),
    Color3.fromRGB(100, 180, 255)
)

AutoBtn.MouseButton1Click:Connect(function()
    if autoRunning then
        autoRunning = false
        AutoLbl.Text       = "Continuous Click: OFF"
        AutoLbl.TextColor3 = Color3.fromRGB(100, 180, 255)
        StatusLbl.Text     = "  auto-click stopped"
        return
    end

    local query = SearchBox.Text
    if query == "" then
        StatusLbl.Text = "  type a name first"
        return
    end

    autoRunning        = true
    AutoLbl.Text       = "Continuous Click: ON  ◼ stop"
    AutoLbl.TextColor3 = Color3.fromRGB(255, 100, 100)

    task.spawn(function()
        while autoRunning do
            -- Fresh search every cycle so newly spawned UI elements are caught
            local count = RunSearch(query)
            BuildResultRows(lastResults)

            if not count or count == 0 then
                StatusLbl.Text = "  no results, retrying..."
                task.wait(PersistState.ClickDelay)
            else
                for i, entry in ipairs(lastResults) do
                    if not autoRunning then break end

                    local inst = entry.instance
                    if inst and inst.Parent then
                        local fired = FireElement(inst)
                        StatusLbl.Text = "  [" .. i .. "/" .. count .. "] "
                            .. (fired and "" or "(no method) ")
                            .. entry.name
                    end

                    task.wait(PersistState.ClickDelay)
                end
            end
        end

        AutoLbl.Text       = "Continuous Click: OFF"
        AutoLbl.TextColor3 = Color3.fromRGB(100, 180, 255)
    end)
end)

end)

-- ─── DRAG ────────────────────────────────────────────────────
local dragging, dragInput, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging  = true
        dragStart = input.Position
        startPos  = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

-- ─── DEATH PERSISTENCE ───────────────────────────────────────
-- UiScanner doesn't depend on character, but stop auto-click on death
-- and resume after respawn so it doesn't click on a stale list
local wasAutoRunning = false

LocalPlayer.CharacterAdded:Connect(function()
    if autoRunning then
        wasAutoRunning = true
        autoRunning    = false
    end
    task.delay(1, function()
        if wasAutoRunning then
            wasAutoRunning = false
            -- Re-trigger the auto-click button as if pressed
            AutoBtn.MouseButton1Click:Fire()
        end
    end)
end)
