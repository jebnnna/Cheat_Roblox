local function GetHRP()
    Character        = LocalPlayer.Character
    HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")
    return HumanoidRootPart
end

local function DistanceTo(inst)
    local root = GetHRP()
    if not root then return 0 end
    local pos
    if inst:IsA("BasePart") then
        pos = inst.Position
    elseif inst:IsA("Model") then
        local p = inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart")
        if p then pos = p.Position end
    end
    if not pos then return 0 end
    return math.floor((root.Position - pos).Magnitude)
end

-- RunSearch respects FilterMode
local function RunSearch(query)
    ClearHighlights()
    lastResults = {}
    if query == "" then
        StatusLbl.Text = "  type a name and press Search"
        BuildResultRows({})
        return
    end

    local queryLower = query:lower()
    local mode       = PersistState.FilterMode
    hlFolder         = Instance.new("Folder")
    hlFolder.Name    = "Scanner_HL"
    hlFolder.Parent  = workspace

    local seen = {}

    local function PassesFilter(inst)
        if mode == "All" then
            return inst:IsA("BasePart") or inst:IsA("Model")
        elseif mode == "Parts" then
            return inst:IsA("BasePart")
        elseif mode == "Models" then
            return inst:IsA("Model")
        end
        return false
    end

    local function ScanDescendants(parent)
        for _, inst in ipairs(parent:GetDescendants()) do
            if seen[inst] then continue end
            if not inst.Name:lower():find(queryLower, 1, true) then continue end
            if not PassesFilter(inst) then continue end

            seen[inst] = true
            local dist = DistanceTo(inst)

            local hl = Instance.new("Highlight")
            hl.FillColor           = Color3.fromRGB(200, 100, 255)
            hl.OutlineColor        = Color3.fromRGB(255, 255, 255)
            hl.FillTransparency    = 0.5
            hl.OutlineTransparency = 0
            hl.DepthMode           = Enum.HighlightDepthMode.AlwaysOnTop
            hl.Adornee             = inst
            hl.Parent              = hlFolder

            local labelBase
            if inst:IsA("BasePart") then
                labelBase = inst
            elseif inst:IsA("Model") then
                labelBase = inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart")
            end
            if labelBase then
                local bb = Instance.new("BillboardGui")
                bb.Size        = UDim2.new(0, 140, 0, 28)
                bb.StudsOffset = Vector3.new(0, 4, 0)
                bb.AlwaysOnTop = true
                bb.Adornee     = labelBase
                bb.Parent      = hlFolder
                local lbl = Instance.new("TextLabel")
                lbl.Text                   = inst.Name
                lbl.Size                   = UDim2.new(1, 0, 1, 0)
                lbl.BackgroundTransparency = 1
                lbl.TextColor3             = Color3.fromRGB(200, 100, 255)
                lbl.TextStrokeTransparency = 0
                lbl.TextStrokeColor3       = Color3.fromRGB(0, 0, 0)
                lbl.Font                   = Enum.Font.GothamBold
                lbl.TextSize               = 13
                lbl.Parent                 = bb
            end

            lastResults[#lastResults + 1] = {
                instance = inst,
                name     = inst.Name,
                class    = inst.ClassName,
                dist     = dist,
            }
        end
    end

    ScanDescendants(workspace)
    table.sort(lastResults, function(a, b) return a.dist < b.dist end)

    local count = #lastResults
    StatusLbl.Text = "  " .. count .. " result" .. (count == 1 and "" or "s")
        .. " (" .. mode .. ") for: " .. query
    BuildResultRows(lastResults)
end

-- The radar scan — called on Heartbeat, rebuilds highlights every ~0.5s
local radarTick = 0

local function RunRadarScan()
    local root = GetHRP()
    if not root then return end

    ClearRadarHighlights()
    radarFolder        = Instance.new("Folder")
    radarFolder.Name   = "Scanner_Radar"
    radarFolder.Parent = workspace

    local radius   = PersistState.RadarRadius
    local myPos    = root.Position
    local count    = 0
    local seen     = {}

    for _, inst in ipairs(workspace:GetDescendants()) do
        if seen[inst] then continue end
        -- skip character parts
        if Character and inst:IsDescendantOf(Character) then continue end
        -- skip existing scanner GUI folders
        if inst == hlFolder or inst == radarFolder then continue end

        local pos
        if inst:IsA("BasePart") then
            pos = inst.Position
        elseif inst:IsA("Model") then
            if inst.PrimaryPart then
                pos = inst.PrimaryPart.Position
            else
                local p = inst:FindFirstChildWhichIsA("BasePart")
                if p then pos = p.Position end
            end
        end

        if not pos then continue end
        local dist = (myPos - pos).Magnitude
        if dist > radius then continue end

        -- Avoid double-highlighting child parts when model is already tagged
        seen[inst] = true

        local hl = Instance.new("Highlight")
        hl.FillColor           = Color3.fromRGB(80, 200, 255)
        hl.OutlineColor        = Color3.fromRGB(255, 255, 255)
        hl.FillTransparency    = 0.55
        hl.OutlineTransparency = 0
        hl.DepthMode           = Enum.HighlightDepthMode.AlwaysOnTop
        hl.Adornee             = inst
        hl.Parent              = radarFolder

        local labelBase
        if inst:IsA("BasePart") then
            labelBase = inst
        elseif inst:IsA("Model") then
            labelBase = inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart")
        end

        if labelBase then
            local bb = Instance.new("BillboardGui")
            bb.Size        = UDim2.new(0, 160, 0, 28)
            bb.StudsOffset = Vector3.new(0, 4, 0)
            bb.AlwaysOnTop = true
            bb.Adornee     = labelBase
            bb.Parent      = radarFolder

            local lbl = Instance.new("TextLabel")
            lbl.Text                  = inst.Name .. " [" .. math.floor(dist) .. "st]"
            lbl.Size                  = UDim2.new(1, 0, 1, 0)
            lbl.BackgroundTransparency= 1
            lbl.TextColor3            = Color3.fromRGB(80, 200, 255)
            lbl.TextStrokeTransparency= 0
            lbl.TextStrokeColor3      = Color3.fromRGB(0, 0, 0)
            lbl.Font                  = Enum.Font.GothamBold
            lbl.TextSize              = 12
            lbl.Parent                = bb
        end

        count = count + 1
    end

    RadarStatusLbl.Text = "  " .. count .. " object" .. (count == 1 and "" or "s") .. " within " .. radius .. " studs"
end

local function StartRadar()
    PersistState.RadarActive = true
    RadarBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
    RadarLbl.Text             = "Radar: ON"
    RadarLbl.TextColor3       = Color3.fromRGB(80, 200, 255)

    -- Disconnect old connection safely
    if radarConn then radarConn:Disconnect() radarConn = nil end

    -- Run immediately, then every 0.5s via Heartbeat accumulator
    RunRadarScan()
    radarConn = RunService.Heartbeat:Connect(function(dt)
        radarTick = radarTick + dt
        if radarTick >= 0.5 then
            radarTick = 0
            RunRadarScan()
        end
    end)
end

local function StopRadar()
    PersistState.RadarActive = false
    if radarConn then radarConn:Disconnect() radarConn = nil end
    ClearRadarHighlights()
    RadarBtn.BackgroundColor3 = Color3.fromRGB(25, 20, 40)
    RadarLbl.Text             = "Radar: OFF"
    RadarLbl.TextColor3       = Color3.fromRGB(210, 210, 210)
    RadarStatusLbl.Text       = "  radar off"
end

RadarBtn.MouseButton1Click:Connect(function()
    if PersistState.RadarActive then
        StopRadar()
    else
        StartRadar()
    end
end)

-- ── Continuous Teleport ───────────────────────────────────────
-- Loops through all search results, teleporting one by one.
-- When the list runs out it re-runs the search automatically and loops again.
-- Second press stops it.
local contTpRunning = false

local ContTpBtn, ContTpLbl = CreateActionButton(
    "Continuous Teleport: OFF",
    Color3.fromRGB(20, 30, 45),
    Color3.fromRGB(40, 80, 120),
    Color3.fromRGB(100, 180, 255)
)

ContTpBtn.MouseButton1Click:Connect(function()
    if contTpRunning then
        contTpRunning = false
        ContTpLbl.Text       = "Continuous Teleport: OFF"
        ContTpLbl.TextColor3 = Color3.fromRGB(100, 180, 255)
        StatusLbl.Text       = "  continuous teleport stopped"
        return
    end

    local query = SearchBox.Text
    if query == "" then
        StatusLbl.Text = "  type a name first"
        return
    end

    contTpRunning        = true
    ContTpLbl.Text       = "Continuous Teleport: ON "
    ContTpLbl.TextColor3 = Color3.fromRGB(255, 100, 100)

    task.spawn(function()
        while contTpRunning do
            -- Re-run search to get fresh list (respects current filter)
            RunSearch(query)

            if #lastResults == 0 then
                -- Nothing found — wait a bit and retry
                StatusLbl.Text = "  no results, retrying..."
                task.wait(PersistState.TpDelay)
            else
                -- Walk the list
                for i, entry in ipairs(lastResults) do
                    if not contTpRunning then break end

                    local root = GetHRP()
                    if not root then
                        contTpRunning = false
                        break
                    end

                    local inst = entry.instance
                    if not inst or not inst.Parent then
                        -- Instance removed mid-loop; skip silently
                    else
                        local pos
                        if inst:IsA("BasePart") then
                            pos = inst.Position
                        elseif inst:IsA("Model") then
                            local p = inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart")
                            if p then pos = p.Position end
                        end
                        if pos then
                            root.CFrame    = CFrame.new(pos) * CFrame.new(0, 4, 0)
                            StatusLbl.Text = "  [loop " .. i .. "/" .. #lastResults .. "] " .. entry.name
                        end
                    end

                    task.wait(PersistState.TpDelay)
                end
                -- List exhausted — loop will re-run RunSearch at top of while
            end
        end

        -- Cleanup after stop
        ContTpLbl.Text       = "Continuous Teleport: OFF"
        ContTpLbl.TextColor3 = Color3.fromRGB(100, 180, 255)
    end)
end)

-- ── Add Radar Detections to List ─────────────────────────────
local AddRadarBtn, AddRadarLbl = CreateActionButton(
    "Add Radar Detections to List",
    Color3.fromRGB(20, 35, 20),
    Color3.fromRGB(40, 100, 40),
    Color3.fromRGB(100, 220, 100)
)

AddRadarBtn.MouseButton1Click:Connect(function()
    local root = GetHRP()
    if not root then
        StatusLbl.Text = "  no character"
        return
    end

    local radius  = PersistState.RadarRadius
    local myPos   = root.Position
    local added   = 0
    local existing = {}

    for _, e in ipairs(lastResults) do
        existing[e.instance] = true
    end

    for _, inst in ipairs(workspace:GetDescendants()) do
        if Character and inst:IsDescendantOf(Character) then continue end
        if not (inst:IsA("BasePart") or inst:IsA("Model")) then continue end

        local pos
        if inst:IsA("BasePart") then
            pos = inst.Position
        elseif inst:IsA("Model") then
            local p = inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart")
            if p then pos = p.Position end
        end

        if not pos then continue end
        if (myPos - pos).Magnitude > radius then continue end
        if existing[inst] then continue end

        existing[inst] = true
        lastResults[#lastResults + 1] = {
            instance = inst,
            name     = inst.Name,
            class    = inst.ClassName,
            dist     = math.floor((myPos - pos).Magnitude),
        }
        added = added + 1
    end

    table.sort(lastResults, function(a, b) return a.dist < b.dist end)
    BuildResultRows(lastResults)
    StatusLbl.Text = "  added " .. added .. " object" .. (added == 1 and "" or "s") .. " (" .. #lastResults .. " total)"
end)

local dragging, dragInput, dragStart, startPos

        dragging  = true
        dragStart = input.Position
        startPos  = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

-- ─── DEATH PERSISTENCE ───────────────────────────────────────
-- On death: re-anchor HRP reference, re-apply radar if it was active.
-- PersistState is never wiped — radar keeps running through spawns.
LocalPlayer.CharacterAdded:Connect(function(char)
    Character        = char
    HumanoidRootPart = char:WaitForChild("HumanoidRootPart")

    -- Radar was active before death → restart it on the new character
    if PersistState.RadarActive then
        -- Small delay so the character settles into the world
        task.delay(0.5, function()
            StartRadar()
        end)
    end
end)

-- Restore radar state on initial load if it was somehow already true
if PersistState.RadarActive then
    StartRadar()
end
