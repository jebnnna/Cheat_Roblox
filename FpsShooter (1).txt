local function CreateToggle(name, state_key, callback)

    BtnStroke.Thickness = 1
    BtnStroke.Parent    = Btn

    local Dot = Instance.new("Frame")
    Dot.Size             = UDim2.new(0, 10, 0, 10)
    Dot.Position         = UDim2.new(0, 14, 0.5, -5)
    Dot.BackgroundColor3 = Color3.fromRGB(80, 80, 80)
    Dot.BorderSizePixel  = 0
    Dot.Parent           = Btn
    Instance.new("UICorner", Dot).CornerRadius = UDim.new(1, 0)

    local BtnLabel = Instance.new("TextLabel")
    BtnLabel.Text              = name
    BtnLabel.Size              = UDim2.new(1, -36, 1, 0)
    BtnLabel.Position          = UDim2.new(0, 32, 0, 0)
    BtnLabel.BackgroundTransparency = 1
    BtnLabel.TextColor3        = Color3.fromRGB(200, 200, 220)
    BtnLabel.TextSize          = 13
    BtnLabel.Font              = Enum.Font.Gotham
    BtnLabel.TextXAlignment    = Enum.TextXAlignment.Left
    BtnLabel.Parent            = Btn

    local StatusLabel = Instance.new("TextLabel")
    StatusLabel.Text              = "OFF"
    StatusLabel.Size              = UDim2.new(0, 40, 1, 0)
    StatusLabel.Position          = UDim2.new(1, -46, 0, 0)
    StatusLabel.BackgroundTransparency = 1
    StatusLabel.TextColor3        = Color3.fromRGB(120, 80, 160)
    StatusLabel.TextSize          = 11
    StatusLabel.Font              = Enum.Font.GothamBold
    StatusLabel.Parent            = Btn

    local function UpdateVisual()
        if State[state_key] then
            Dot.BackgroundColor3  = Color3.fromRGB(230, 230, 230)
            BtnStroke.Color       = Color3.fromRGB(210, 210, 210)
            Btn.BackgroundColor3  = Color3.fromRGB(35, 35, 35)
            StatusLabel.Text      = "ON"
            StatusLabel.TextColor3= Color3.fromRGB(230, 230, 230)
        else
            Dot.BackgroundColor3  = Color3.fromRGB(60, 60, 70)
            BtnStroke.Color       = Color3.fromRGB(55, 55, 55)
            Btn.BackgroundColor3  = Color3.fromRGB(25, 20, 40)
            StatusLabel.Text      = "OFF"
            StatusLabel.TextColor3= Color3.fromRGB(120, 120, 120)
        end
    end

    Btn.MouseButton1Click:Connect(function()
        State[state_key] = not State[state_key]
        UpdateVisual()
        if callback then callback(State[state_key]) end
    end)

    UpdateVisual()
    ToggleRegistry[state_key] = UpdateVisual
    return Btn
end

-- ─── AIMBOT CORE ─────────────────────────────────────────────
local function GetClosestTarget()
    local vCenter  = Camera.ViewportSize / 2
    local bestDist = State.AimbotFOV
    local bestBone = nil
    local boneName = State.AimHead and "Head" or "HumanoidRootPart"
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer or not plr.Character then continue end
        if State.TeamCheck and plr.Team == LocalPlayer.Team then continue end
        local char = plr.Character
        local hum  = char:FindFirstChildOfClass("Humanoid")
        local bone = char:FindFirstChild(boneName)
        if hum and hum.Health > 0 and bone then
            local screenPos, onScreen = Camera:WorldToViewportPoint(bone.Position)
            if onScreen then
                local dist = (Vector2.new(screenPos.X, screenPos.Y) - vCenter).Magnitude
                if dist < bestDist then
                    bestDist = dist
                    bestBone = bone
                end
            end
        end
    end
    return bestBone
end

RunService.RenderStepped:Connect(function()
    if not State.Aimbot then return end
    local target = GetClosestTarget()
    if target then
        local goal = CFrame.new(Camera.CFrame.Position, target.Position)
        Camera.CFrame = Camera.CFrame:Lerp(goal, State.AimbotSmooth)
    end
end)

-- ─── ESP CORE ────────────────────────────────────────────────
local espFolder, hlFolder

local function ClearESP()
    if espFolder then espFolder:Destroy() espFolder = nil end
end
local function ClearHL()
    if hlFolder then hlFolder:Destroy() hlFolder = nil end
end

local function BuildESP()
    ClearESP()
    espFolder = Instance.new("Folder")
    espFolder.Name   = "FPS_ESP"
    espFolder.Parent = workspace
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local root = player.Character:FindFirstChild("HumanoidRootPart")
            if root then
                local bb = Instance.new("BillboardGui")
                bb.Name        = "ESP_" .. player.Name
                bb.Size        = UDim2.new(0, 120, 0, 30)
                bb.StudsOffset = Vector3.new(0, 3, 0)
                bb.AlwaysOnTop = true
                bb.Adornee     = root
                bb.Parent      = espFolder
                local glow = Instance.new("TextLabel")
                glow.Text = player.Name
                glow.Size = UDim2.new(1, 0, 1, 0)
                glow.BackgroundTransparency = 1
                glow.TextColor3 = Color3.fromRGB(255, 255, 255)
                glow.TextTransparency = 0.6
                glow.TextStrokeTransparency = 0
                glow.Font = Enum.Font.GothamBold
                glow.TextSize = 15
                glow.ZIndex = 0
                glow.Parent = bb
                local lbl = Instance.new("TextLabel")
                lbl.Text = player.Name
                lbl.Size = UDim2.new(1, 0, 1, 0)
                lbl.BackgroundTransparency = 1
                lbl.TextColor3 = Color3.fromRGB(200, 100, 255)
                lbl.TextStrokeTransparency = 0
                lbl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                lbl.Font = Enum.Font.GothamBold
                lbl.TextSize = 13
                lbl.ZIndex = 1
                lbl.Parent = bb
            end
        end
    end
end

local function BuildHL()
    ClearHL()
    hlFolder = Instance.new("Folder")
    hlFolder.Name   = "FPS_HL"
    hlFolder.Parent = workspace
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local hl = Instance.new("Highlight")
            hl.FillColor          = Color3.fromRGB(200, 100, 255)
            hl.OutlineColor       = Color3.fromRGB(255, 255, 255)
            hl.FillTransparency   = 0.6
            hl.OutlineTransparency= 0
            hl.DepthMode          = Enum.HighlightDepthMode.AlwaysOnTop
            hl.Adornee            = player.Character
            hl.Parent             = hlFolder
        end
    end
end

Players.PlayerAdded:Connect(function()
    if State.ESP then BuildESP() end
    if State.ESPGlow then BuildHL() end
end)
Players.PlayerRemoving:Connect(function()
    if State.ESP then BuildESP() end
    if State.ESPGlow then BuildHL() end
end)

-- ─── DRAG ─────────────────────────────────────────────────────
local dragging, dragInput, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging  = true
        dragStart = input.Position
        startPos  = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)
